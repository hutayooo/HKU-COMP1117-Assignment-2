# Function checks

# 1. Refused ride resets discount (handled in program state) & general fare check
ride_fare(1, 10.0, False) == 10.0

# 2. Top-up landing on maximum cap (3000)
can_top_up(2500.0, 500.0) == True
can_top_up(2500.0, 501.0) == False

# 3. Ride balance limit edge at -35
can_ride(0.0, 35.0) == False
can_ride(1.0, 36.0) == True
can_ride(1.0, 36.5) == False

# 4. Pre-ride balance strictly greater than 0
can_ride(0.0, 5.0) == False
can_ride(-5.0, 5.0) == False

# 5. Top-up extremes (50 and 1000)
can_top_up(100.0, 50.0) == True
can_top_up(100.0, 1000.0) == True
can_top_up(100.0, 49.0) == False
can_top_up(100.0, 1001.0) == False

# 6. Non-multiple of 50 top-up
can_top_up(100.0, 75.0) == False

# 7. Fare floor at 0 after interchange discount
ride_fare(1, 0.5, True) == 0.0

# 8. Elder fare boundary at 2 cap vs below cap
ride_fare(3, 3.0, False) == 2.0
ride_fare(3, 1.0, False) == 1.0
